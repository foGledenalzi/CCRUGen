import { regionLabel, type RegionId } from '../regions'
export const PAGE_STEP = 10            // UI-SPEC [default]: Page Up/Down jump 10 zones
export type RingTarget =
  | { readonly kind: 'zone'; readonly zone: number }
  | { readonly kind: 'syzygy'; readonly zone: number }
  | { readonly kind: 'current'; readonly zone: number }
  | { readonly kind: 'gate'; readonly zone: number }
  | { readonly kind: 'pair'; readonly pair: number }
export interface KeyboardContext {
  readonly mode: 'zones' | 'pairs'
  readonly order: readonly number[]        // visible zones (zones mode) or visible pair ids (pairs mode), ascending; muted excluded
  hasSyzygy(zone: number): boolean          // drawn and not muted (layer on, both ends visible)
  hasCurrent(zone: number): boolean
  hasGate(zone: number): boolean
}
export type KeyOutcome =
  | { readonly kind: 'move'; readonly target: RingTarget }
  | { readonly kind: 'activate'; readonly target: RingTarget }
  | { readonly kind: 'zoom'; readonly action: 'in' | 'out' | 'fit' }
  | { readonly kind: 'leave' }               // do not preventDefault: the browser moves focus on
  | { readonly kind: 'ignore' }
export function initialTarget(ctx: KeyboardContext): RingTarget | null
export function nextTarget(current: RingTarget | null, key: string, shiftKey: boolean, ctx: KeyboardContext): KeyOutcome
export interface TargetText {
  readonly zoneLabels: readonly string[]
  readonly zoneRegion: readonly RegionId[]
  syzygyOf(zone: number): { readonly a: number; readonly b: number } | null     // the view.syzygies entry of the zone's pair
  currentOf(zone: number): { readonly name: string; readonly label: string } | null
  gateOf(zone: number): { readonly name: string; readonly from: number; readonly to: number } | null
  pairOf(q: number): { readonly lo: number; readonly hi: number }
  isSelected(zone: number): boolean
}
export function focusKeyOf(target: RingTarget, text: TargetText): string
export function announcementFor(target: RingTarget, text: TargetText): string
export function activationAnnouncement(target: RingTarget, text: TargetText, nowSelected: boolean): string
```

export const CLICK_SLOP_PX = 3   // mouse/pen: release within this many CSS px of the press is a click (DemonMatrix DRAG_THRESHOLD_PX)
export const TOUCH_SLOP_PX = 8   // touch: [default] pan threshold and tap slop
export type PointerKind = 'mouse' | 'pen' | 'touch'
export interface PointerInput {
  readonly type: 'down' | 'move' | 'up' | 'cancel' | 'leave'
  readonly pointerId: number
  readonly pointerType: PointerKind
  readonly x: number            // CSS px relative to the canvas
  readonly y: number
  readonly button: number
  readonly altKey: boolean
  readonly shiftKey: boolean
  readonly overTarget: boolean  // the component's pick at this point hit something (only read on 'down')
}
export interface TrackedPointer { readonly x: number; readonly y: number; readonly startX: number; readonly startY: number; readonly kind: PointerKind }
export interface PointerState {
  readonly mode: 'idle' | 'press' | 'pan' | 'pinch' | 'rubber'
  readonly pointers: ReadonlyMap<number, TrackedPointer>
  readonly pinchDistance: number
  readonly pinchMidX: number
  readonly pinchMidY: number
  readonly gesture: boolean     // gestureStart emitted and not yet ended
}
export type PointerAction =
  | { readonly kind: 'hover'; readonly x: number; readonly y: number }
  | { readonly kind: 'hoverEnd' }
  | { readonly kind: 'click'; readonly x: number; readonly y: number }
  | { readonly kind: 'tap'; readonly x: number; readonly y: number }
  | { readonly kind: 'pan'; readonly dx: number; readonly dy: number }
  | { readonly kind: 'zoom'; readonly factor: number; readonly x: number; readonly y: number }
  | { readonly kind: 'gestureStart' }
  | { readonly kind: 'gestureEnd' }
  | { readonly kind: 'rubberStart'; readonly x: number; readonly y: number; readonly additive: boolean }
  | { readonly kind: 'capture'; readonly pointerId: number }
export const INITIAL_POINTER_STATE: PointerState
export function reducePointer(state: PointerState, input: PointerInput): { readonly state: PointerState; readonly actions: readonly PointerAction[] }
export function wheelZoomFactor(deltaY: number, deltaMode: number): number
```

```typescript
import { DIAGRAM_CSS_WIDTH } from '../viewLayouts'
export const ZOOM_MIN = 0.3

export interface CanvasTransform { readonly zoom: number; readonly tx: number; readonly ty: number }   // tx/ty in CSS px
export const IDENTITY_TRANSFORM: CanvasTransform
export function fitScale(frame: CanvasFrame, w: number, h: number): number
export function scaleOf(t: CanvasTransform, frame: CanvasFrame, w: number, h: number): number          // fitScale * zoom
export function onScreenRadius(t: CanvasTransform, frame: CanvasFrame, w: number, h: number): number    // frame.nodeRadius * scaleOf
export function zoomMax(frame: CanvasFrame, w: number, h: number): number
export function toScreen(t: CanvasTransform, frame: CanvasFrame, w: number, h: number, wx: number, wy: number): { x: number; y: number }
export function toWorld(t: CanvasTransform, frame: CanvasFrame, w: number, h: number, sx: number, sy: number): { x: number; y: number }
export function clampTransform(t: CanvasTransform, frame: CanvasFrame, w: number, h: number): CanvasTransform
export function zoomAt(t: CanvasTransform, factor: number, sx: number, sy: number, frame: CanvasFrame, w: number, h: number): CanvasTransform
export function panBy(t: CanvasTransform, dx: number, dy: number, frame: CanvasFrame, w: number, h: number): CanvasTransform

export function ensureVisible(t: CanvasTransform, wx: number, wy: number, radiusPx: number, frame: CanvasFrame, w: number, h: number): CanvasTransform
export function canvasFootprint(frame: CanvasFrame, viewportW: number, viewportH: number): { cssWidth: number; cssHeight: number }
export function backingStore(cssW: number, cssH: number, devicePixelRatio: number, areaLimitPx?: number): { dpr: number; pixelWidth: number; pixelHeight: number }
```

backingStore: `dpr = Math.min(DPR_MAX, finite positive devicePixelRatio or 1)`; if `cssW * cssH * dpr * dpr > limit` then `dpr = Math.sqrt(limit / (cssW * cssH))`; `pixelWidth = Math.floor(cssW * dpr)`, `pixelHeight = Math.floor(cssH * dpr)`.

```typescript
import { PICK_FLATTEN_STEPS, flattenQuad, parseRouteCurve, type CanvasScene } from '../../../engine/index'
import { curveAway } from '../geometry'
import type { Demon } from '../../data/types'
export const PICK_RADIUS_PX = 12          // D-09: minimum hit radius in CSS px at any zoom (a 24 px WCAG 2.5.8 target)
export const TIE_ORDER = ['zone', 'syzygy', 'current', 'gate', 'chord'] as const
export type PickKind = 'zone' | 'pair' | 'syzygy' | 'current' | 'gate' | 'chord'
export interface PickHit { readonly kind: PickKind; readonly id: number; readonly distancePx: number }   // id: zone, pair id, pair id, pair id, origin zone, chord index
export interface ZoneGrid { readonly cellSize: number; readonly minX: number; readonly minY: number; readonly cols: number; readonly rows: number; readonly cellStart: Int32Array; readonly items: Int32Array }
export interface PickFilter { zoneAllowed(zone: number): boolean; edgeAllowed(kind: 'syzygy' | 'current' | 'gate', id: number): boolean }
export interface FlatChords { readonly segments: Float64Array; readonly chordStart: Int32Array }   // CSR over chords, 4 numbers per segment
export function buildZoneGrid(x: Float64Array, y: Float64Array, nodeRadius: number): ZoneGrid     // cell = max(4 * nodeRadius, 1e-6); CSR arrays
export function pickAt(scene: CanvasScene, grid: ZoneGrid, wx: number, wy: number, scalePx: number, filter: PickFilter, chords?: FlatChords | null): PickHit | null
export function zonesInWorldRect(scene: CanvasScene, minX: number, minY: number, maxX: number, maxY: number, filter: PickFilter): number[]
export function distToSegment(px: number, py: number, x0: number, y0: number, x1: number, y1: number): number
export function distToStadium(px: number, py: number, cx: number, cy: number, width: number, height: number): number   // capsule of the pill (rx = height / 2)
export function flattenChords(chords: readonly Demon[], x: Float64Array, y: Float64Array, cx: number, cy: number): FlatChords   // curveAway(..., 0.25) parsed with parseRouteCurve, flattened with flattenQuad
```
pickAt: `radiusWorld = PICK_RADIUS_PX / scalePx`; zones (kind 'zones'): query grid cells overlapping `[w - (R + radiusWorld), w + (R + radiusWorld)]`, metric `Math.max(0, hypot * scalePx - R * scalePx)`;

chords: min distance per chord. Keep only candidates with distancePx <= PICK_RADIUS_PX; nearest wins; ties by TIE_ORDER rank ('pair' as 'zone') then lower id.

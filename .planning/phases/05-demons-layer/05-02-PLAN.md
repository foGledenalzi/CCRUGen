```typescript
export const MATRIX_MAX_CELL_PX = 64
export function fitScale(base: number, w: number, h: number): number        // Math.max(1e-12, (Math.min(w, h) - 2 * MATRIX_PAD_PX) / base); returns 0-safe value for w/h <= 16
export function fitTransform(base: number, w: number, h: number): MatrixTransform  // scale = fitScale; tx = (w - base*scale)/2; ty = (h - base*scale)/2
export function clampTransform(t: MatrixTransform, base: number, w: number, h: number): MatrixTransform
export function zoomAt(t: MatrixTransform, factor: number, px: number, py: number, base: number, w: number, h: number): MatrixTransform
export function panBy(t: MatrixTransform, dx: number, dy: number, base: number, w: number, h: number): MatrixTransform
export function cellAtPixel(px: number, py: number, t: MatrixTransform, base: number): Cell | null
export function cellCenter(a: number, b: number, t: MatrixTransform): { readonly x: number; readonly y: number }
export function cellRect(a: number, b: number, t: MatrixTransform): { readonly x: number; readonly y: number; readonly size: number }
export type CursorKey = 'ArrowLeft' | 'ArrowRight' | 'ArrowUp' | 'ArrowDown'
export function stepCursor(cell: Cell, key: CursorKey, base: number): Cell
export function ensureCellVisible(t: MatrixTransform, a: number, b: number, base: number, w: number, h: number): MatrixTransform
export function syzygyLine(t: MatrixTransform, base: number): Segment          // cells a + b = base - 1, a > b: from (base/2, base/2 - 1) to (base - 1, 0), through cell centres
export function numodemonLine(t: MatrixTransform, base: number): Segment | null // cells a + b = base, a > b: from (base/2 + 1, base/2 - 1) to (base - 1, 1); null when base < 4
```
Rules:
- clampTransform: scale clamped to [fitScale, Math.max(MATRIX_MAX_CELL_PX, fitScale)]; then tx clamped to [w/2 - base*scale, w/2] and ty to
  [h/2 - base*scale, h/2] (the viewport centre always lies inside the matrix square). Any non-finite input field falls back to fitTransform (T-05-06).
- zoomAt: s2 = clamped(t.scale * factor); tx2 = px - (px - t.tx) * (s2 / t.scale); ty2 likewise; then clampTransform. factor <= 0 or non-finite returns t unchanged.
- panBy: clampTransform({ scale, tx: tx + dx, ty: ty + dy }).
- cellAtPixel: if any input is non-finite return null; a = Math.floor((px - tx) / scale), b = Math.floor((py - ty) / scale); valid only when
  0 <= b && b < a && a < base. This function is the ONLY way hover, click and keyboard resolve a cell — never from raster pixels (T-05-05).
- stepCursor moves one step and returns the input cell when the step would leave the triangle (Left needs a - 1 > b; Right needs a + 1 < base; Up needs b - 1 >= 0; Down needs b + 1 < a).
- ensureCellVisible: shift tx/ty by the minimum amount that puts cellRect inside [0, w] x [0, h], then clampTransform.
Doubles are exact for every coordinate here (base * 64 <= 2^32 << 2^53), so no BigInt. Keep the base-ten grep gate green (no `<= 4` literal comparisons:
write numodemonLine's guard as `base < 4`).

```typescript
export const MATRIX_MAX_RASTER_PX = 1_500_000
export const MATRIX_MAX_DPR = 2
export const MATRIX_KIND_ORDER = ['chrono', 'amphi', 'xeno', 'syzygy'] as const
export const MATRIX_BG = '#08080f'     // UI-SPEC secondary rgba(8,8,15): outside the matrix square
export const MATRIX_EMPTY = '#0d0d16'  // inside the square, b >= a (no demon)
export const PALETTE_BG = 4
export const PALETTE_EMPTY = 5
export function rasterSize(cssW: number, cssH: number, dpr: number): { readonly width: number; readonly height: number; readonly ratio: number }
export function buildPalette(hexes: readonly string[]): Uint8ClampedArray   // '#rrggbb' -> [r, g, b, 255]; RangeError on anything else
export interface RasterJob {
  readonly t: MatrixTransform          // CSS-px transform the raster is committed to
  readonly base: number
  readonly width: number               // backing pixels
  readonly height: number
  readonly ratio: number               // backing px per CSS px
  readonly classify: (a: number, b: number) => number   // palette index 0..3 for a valid cell a > b
  readonly palette: Uint8ClampedArray  // >= 6 RGBA entries
  readonly out: Uint8ClampedArray      // width * height * 4 bytes (ImageData.data)
}
export function rasterizeRows(job: RasterJob, rowStart: number, rowEnd: number): number  // returns classify calls made
```

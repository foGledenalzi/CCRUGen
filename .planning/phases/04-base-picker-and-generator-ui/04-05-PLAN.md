  readonly region: RegionId | null; readonly tc: boolean; readonly particles: boolean; readonly date: string; readonly orbits: boolean
  readonly labels: LabelScheme; readonly isolate: readonly RegionId[]; readonly mute: readonly RegionId[]; readonly packer: Packer; readonly tier: RenderTier | null

export interface BaseRefusal { readonly raw: string; readonly check: BaseCheck | null }   // check null = not a plain decimal integer

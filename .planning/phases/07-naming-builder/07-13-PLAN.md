[default] decisions recorded here (review at the final checkpoint):
- A link whose link form equals the saved slot's opens the saved slot (it carries the visitor's edits on top of the same seed), not the bare link naming.
- `snapshotNaming` returns the SAVED naming while previewing, so "Keep this naming" is one undoable history entry and "Back to my saved naming" creates none.
- `pending` (panel state) means a worker base switch is in flight past the 150 ms grace; naming generation itself is synchronous (3.3 ms at 4000 zones).
- An undo/redo that restores a naming marks the session edited, so autosave rewrites the slot ("Undo restores the previous in-memory naming and autosave then rewrites the slot").

// app/lib/namingStorage.ts
export interface StorageLike { getItem(key: string): string | null; setItem(key: string, value: string): void }
export function browserStorage(): StorageLike | null            // try { return window.localStorage } catch { return null }; null during SSR
export function slotKey(base: number): string                    // `${NAMING_STORAGE_PREFIX}${base}`
export function readSlot(storage: StorageLike | null, base: number, env: NamingEnv): Naming | null
export function writeSlot(storage: StorageLike | null, naming: Naming): 'saved' | 'failed'
// app/lib/namingSession.ts
export interface NamingEnv { readonly interactiveMaxN: number; readonly preset: NamingPreset | null }
export interface NamingSession {
  readonly base: number; readonly working: Naming; readonly rev: number
  readonly saved: Naming | null; readonly savedRev: number; readonly preview: boolean; readonly edited: boolean
}
export type NamingAction =
  | { readonly type: 'sound'; readonly zone: number; readonly sound: string }
  | { readonly type: 'paste'; readonly zone: number; readonly sounds: readonly string[] }
  | { readonly type: 'lock'; readonly zone: number }
  | { readonly type: 'seed'; readonly seed: string } | { readonly type: 'reroll'; readonly seed: string }
  | { readonly type: 'style'; readonly style: StyleId } | { readonly type: 'join'; readonly join: JoinOrder }
  | { readonly type: 'smoothing'; readonly smoothing: Smoothing } | { readonly type: 'source'; readonly source: 'generated' | 'ccru' }
  | { readonly type: 'override'; readonly mesh: number; readonly name: string } | { readonly type: 'reset-override'; readonly mesh: number }
  | { readonly type: 'import'; readonly naming: Naming } | { readonly type: 'use-seed'; readonly record: SeedRecord }
export type NamesPanelState = 'ready' | 'pending' | 'seed-only' | 'preview'
// app/components/names/NamingContext.tsx
export interface NamingContextValue {
  readonly base: number; readonly naming: Naming; readonly rev: number; readonly resolver: NameResolver
  readonly preset: NamingPreset | null; readonly state: NamesPanelState; readonly pendingBase: number | null
  readonly saved: 'none' | 'saved' | 'failed' | 'preview' | 'seed-only'; readonly uncarried: boolean
  readonly interactiveMaxN: number; readonly announcement: string
  readonly restored: { readonly tick: number; readonly zone: number | null }
  apply(action: NamingAction): boolean; keep(): void; revert(): void; undo(): void; announce(text: string): void
}

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| localStorage -> session | Slots are untrusted input (other tabs, old versions, manual edits) |
| URL link -> session | A shared link must never overwrite a saved slot silently |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-07-08b | Tampering | readSlot | mitigate | Same validator as import; any failure (including a wrong-base or oversized slot) is treated as no slot |
| T-07-09 | Denial of Service | writeSlot / browserStorage | mitigate | try/catch around the getter, getItem and setItem; QuotaExceededError returns 'failed' so the UI shows the "Couldn't save" line |
| T-07-17 | Tampering / Repudiation (silent data loss) | sessionForBase, applyNamingAction | mitigate | Preview when link and slot forms differ; every action refused while previewing; nothing written before the first edit; Keep is undoable |
| T-07-20b | Denial of Service (memory) | snapshotNaming / restoreNaming | mitigate | History holds immutable namings that share unchanged parts; heavy namings get a short undo depth in the client (NAMING_UNDO_HEAVY_DEPTH, plan 07-18) |
</threat_model>
